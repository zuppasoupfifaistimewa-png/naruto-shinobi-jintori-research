def hx(b): return " ".join(f"{x:02X}" for x in b)
def h32(v): return f"0x{v & 0xffffffff:08X}"
def sign16(v): return (v ^ 0x8000) - 0x8000
def reg(r): return "$" + REG[r]
def write_tsv(name, rows, fields):
    with (OUT/name).open("w", encoding="utf-8", newline="") as f:
        w=csv.DictWriter(f, fieldnames=fields, delimiter="\t", extrasaction="ignore", lineterminator="\n")
        w.writeheader()
        for row in rows:
            w.writerow({k: ("TIDAK TAHU" if row.get(k) is None or row.get(k) == "" else row.get(k)) for k in fields})
def write_txt(name, value): (OUT/name).write_text(value, encoding="utf-8")
def decode_blob(blob, tag):
    base, filebase, start = (EXE_BASE, EXE_FILE, EXE_FILE) if tag=="EXE" else (ENTRY_BASE, ENTRY_FILE, 0)
    data=blob[start:]
    out={}
    for p in range(0, len(data)-3, 4):
        raw=data[p:p+4]
        pc=base+p
        ins=next(md.disasm(raw, pc, count=1), None)
        if ins is None: continue
        word=int.from_bytes(raw,"little")
        op=word>>26; rs=(word>>21)&31; rt=(word>>16)&31; rd=(word>>11)&31; sh=(word>>6)&31; fn=word&63
        out[pc]={"pc":pc,"tag":tag,"raw":raw,"word":word,"op":op,"rs":rs,"rt":rt,"rd":rd,"sh":sh,"fn":fn,
                 "imm":sign16(word&0xffff),"uimm":word&0xffff,"mn":ins.mnemonic.lower(),
                 "text":(ins.mnemonic+" "+ins.op_str).strip(),"fileoff":filebase+p}
    return out

def parse_ghidra(path):
    code=set(); ranges=defaultdict(list); refs=defaultdict(list)
    p=Path(path)
    if not p.exists(): return code, ranges, refs
    target=None
    for line in p.read_text(encoding="utf-8", errors="replace").splitlines():
        if line.startswith("CODE_INSTRUCTION="):
            try: code.add(int(line.split("=",1)[1].split("|",1)[0],16))
            except Exception: pass
        elif line.startswith("FUNCTION_RANGE="):
            q=line.split("=",1)[1].split("|")
            if len(q)>=4:
                try: ranges[int(q[0],16)].append((int(q[1],16),int(q[2],16),q[3]))
                except Exception: pass
        elif line.startswith("TARGET="):
            try: target=int(line.split("=",1)[1],16)
            except Exception: target=None
        elif line.startswith("REF=") and target is not None:
            q=line.split("=",1)[1].split("|")
            if len(q)>=4:
                try: refs[target].append({"pc":int(q[0],16),"kind":q[1],"caller_entry":q[2],"caller_name":q[3]})
                except Exception: pass
    return code, ranges, refs

def dest(i):
    op,fn=i["op"],i["fn"]
    if op==0:
        if fn in (8,12,13,24,25,26,27): return None
        if fn==9: return i["rd"] or 31
        return i["rd"] or None
    if op in (2,4,5,6,7,0x28,0x29,0x2b,0x38,0x39,0x3a,0x3b): return None
    if op==3: return 31
    if op in (1,8,9,10,11,12,13,14,15,0x20,0x21,0x22,0x23,0x24,0x25,0x26,0x27,0x30,0x31,0x32,0x33,0x34,0x35,0x36,0x37): return i["rt"] or None
    return None

def direct_target(i):
    if i["op"] not in (2,3): return None
    return ((i["pc"]+4)&0xF0000000) | ((i["word"]&0x03ffffff)<<2)

def last_def(code, r, before, lo, limit=512, stop_control=True):
    floor=max(lo,before-limit*4)
    for pc in range(before-4,floor-1,-4):
        i=code.get(pc)
        if not i: continue
        if dest(i)==r: return i
        if stop_control and (i["op"] in (2,3,4,5,6,7,1) or (i["op"]==0 and i["fn"] in (8,9))):
            return None
    return None

def const_reg(code, r, before, lo, depth=0, seen=None):
    if r==0: return 0
    if depth>12: return None
    seen=set() if seen is None else seen
    key=(r,before)
    if key in seen: return None
    seen.add(key)
    d=last_def(code,r,before,lo)
    if not d: return None
    op,fn=d["op"],d["fn"]
    if op==15: return (d["uimm"]<<16)&0xffffffff
    if op in (8,9,13,12,14):
        a=const_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())
        if a is None: return None
        if op in (8,9): return (a+d["imm"])&0xffffffff
        if op==13: return (a|d["uimm"])&0xffffffff
        if op==12: return (a&d["uimm"])&0xffffffff
        return (a^d["uimm"])&0xffffffff
    if op==0 and fn in (0,2,3,0x21,0x23,0x24,0x25,0x26):
        a=const_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())
        b=const_reg(code,d["rt"],d["pc"],lo,depth+1,seen.copy())
        if fn==0 and b is not None: return (b<<d["sh"])&0xffffffff
        if fn==2 and b is not None: return b>>d["sh"]
        if fn==3 and b is not None: return (b>>d["sh"])
        if a is None or b is None: return None
        if fn==0x21: return (a+b)&0xffffffff
        if fn==0x23: return (a-b)&0xffffffff
        if fn==0x24: return a&b
        if fn==0x25: return a|b
        if fn==0x26: return a^b
    return None

def sym_reg(code, r, before, lo, depth=0, seen=None):
    if r==0: return "0"
    if depth>12: return "TIDAK TAHU:DEPTH"
    seen=set() if seen is None else seen
    key=(r,before)
    if key in seen: return "TIDAK TAHU:CYCLE"
    seen.add(key)
    d=last_def(code,r,before,lo)
    if not d: return "INCOMING_"+reg(r)
    op,fn=d["op"],d["fn"]
    if op==15: return h32(d["uimm"]<<16)
    if op in (8,9): return "("+sym_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())+f"{d['imm']:+d})"
    if op==13: return "("+sym_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())+f"|{h32(d['uimm'])})"
    if op==12: return "("+sym_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())+f"&{h32(d['uimm'])})"
    if op in LOADS:
        a=const_reg(code,d["rs"],d["pc"],lo)
        size={0x20:"8S",0x21:"16S",0x23:"32",0x24:"8",0x25:"16"}[op]
        addr=h32((a+d["imm"])&0xffffffff) if a is not None else sym_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())+f"{d['imm']:+d}"
        return f"MEM{size}[{addr}]"
    if op==0 and fn==0: return "("+sym_reg(code,d["rt"],d["pc"],lo,depth+1,seen.copy())+f"<<{d['sh']})"
    if op==0 and fn in (0x21,0x23,0x24,0x25,0x26):
        if fn==0x21 and d["rs"]==0: return sym_reg(code,d["rt"],d["pc"],lo,depth+1,seen.copy())
        if fn==0x21 and d["rt"]==0: return sym_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())
        opstr={0x21:"+",0x23:"-",0x24:"&",0x25:"|",0x26:"^"}[fn]
        return "("+sym_reg(code,d["rs"],d["pc"],lo,depth+1,seen.copy())+opstr+sym_reg(code,d["rt"],d["pc"],lo,depth+1,seen.copy())+")"
    return f"DEF@{h32(d['pc'])}:{d['text']}"

def eff_addr(code,i,lo):
    b=const_reg(code,i["rs"],i["pc"],lo)
    return ((b+i["imm"])&0xffffffff) if b is not None else None

def base_row(i, excerpt=""):
    z={"image":i["tag"],"pc":h32(i["pc"]),"file_offset":h32(i["fileoff"]),
       "bytes":hx(i["raw"]),"instruksi":i["text"]}
    if excerpt: z["excerpt_id"]=excerpt
    return z

def find_prolog(code, pc, lo, span=0x5000):
    # First accept a prolog exactly at the target, then search backwards but stop at a return.
    def at(q):
        a=code.get(q); b=code.get(q+4)
        if a and b and a["op"]==9 and a["rs"]==29 and a["rt"]==29 and a["imm"]<0 and b["op"]==0x2b and b["rs"]==29 and b["rt"]==31:
            return True
        return False
    if at(pc): return pc,"PROLOG_AT_TARGET"
    floor=max(lo,pc-span)
    for q in range(pc-4,floor-1,-4):
        i=code.get(q)
        if not i: continue
        if i["op"]==0 and i["fn"]==8 and i["rs"]==31: break
        if at(q): return q,"BACKWARD_PROLOG_BEFORE_RETURN"
    return None,"NO_UNIQUE_PROLOG"

def function_for(ranges, pc):
    matches=[]
    for entry,parts in ranges.items():
        for lo,hi,name in parts:
            if lo<=pc<=hi: matches.append((entry,lo,hi,name))
    if not matches:return None
    return sorted(matches,key=lambda x:(x[2]-x[1],x[0]))[0]

def parse_refs(path):
    _,_,refs=parse_ghidra(path)
    return refs

def get_code_excerpt(code, start, count, image, name):
    rows=[]
    for n in range(count):
        i=code.get(start+n*4)
        if i: rows.append(base_row(i,name))
    return rows

def return_sites(code, frange):
    if not frange:return []
    entry,lo,hi,_=frange
    out=[]
    for pc in range(lo,hi+1,4):
        i=code.get(pc)
        if i and i["op"]==0 and i["fn"]==8 and i["rs"]==31:
            out.append({"pc":pc,"value_expr":sym_reg(code,2,pc,lo),"value_const":const_reg(code,2,pc,lo)})
    return out

def call_site_origin(code, ranges, pc, r, manual_entry=None):
    fr=function_for(ranges,pc)
    entry=manual_entry if manual_entry is not None else (fr[0] if fr else None)
    if entry is None:
        return {"origin":"TIDAK TAHU","expression":"TIDAK TAHU: manual prolog/function entry unresolved","definition_pc":"TIDAK TAHU","function_entry":"TIDAK TAHU"}
    delay=code.get(pc+4)
    if delay and dest(delay)==r:
        value=const_reg(code,r,pc+8,entry)
        expr=sym_reg(code,r,pc+8,entry)
        origin="KONSTANTA" if value is not None else ("DARI_MEMORI" if expr.startswith("MEM") else ("ARGUMEN_NASUK" if expr.startswith("INCOMING_") else "TIDAK TAHU"))
        return {"origin":origin,"expression":"DELAY_SLOT "+expr,"definition_pc":h32(delay["pc"]),"function_entry":h32(entry)}
    d=None; crossed_control=False
    for q in range(pc-4,entry-1,-4):
        ins=code.get(q)
        if not ins: continue
        if dest(ins)==r: d=ins; break
        if ins["op"] in (2,3,4,5,6,7,1) or (ins["op"]==0 and ins["fn"] in (8,9)):
            crossed_control=True; break
    if d is None:
        if crossed_control:
            return {"origin":"TIDAK TAHU","expression":"TIDAK TAHU: no reaching definition before control-flow boundary","definition_pc":"TIDAK TAHU","function_entry":h32(entry)}
        return {"origin":"ARGUMEN_MASUK","expression":"INCOMING_"+reg(r),"definition_pc":"TIDAK TAHU","function_entry":h32(entry)}
    value=const_reg(code,r,pc,entry); expr=sym_reg(code,r,pc,entry)
    origin="KONSTANTA" if value is not None else ("DARI_MEMORI" if expr.startswith("MEM") else ("ARGUMEN_MASUK" if expr.startswith("INCOMING_") else "TIDAK TAHU"))
    return {"origin":origin,"expression":expr,"definition_pc":h32(d["pc"]),"function_entry":h32(entry)}

def record_at(blob, pc):
    off=EXE_FILE+(pc-EXE_BASE)
    if off<0 or off+0x1c>len(blob): return None
    raw=blob[off:off+0x1c]
    w=struct.unpack("<7I",raw)
    name_raw=raw[0x14:0x1c]
    try:
        n=name_raw.split(b"\0",1)[0].decode("ascii")
    except Exception: return None
    if w[0]!=0 or w[1]>32 or not n or not re.fullmatch(r"[A-Z0-9_]{1,8}",n): return None
    if not all(0x80010000<=v<=0x801fffff for v in w[2:5]): return None
    return {"base":pc,"plus4":w[1],"p1":w[2],"p2":w[3],"p3":w[4],"name":n,"raw":raw}

def table_scan(exe, code_sets):
    lo,hi=EXE_BASE,EXE_BASE+len(exe)-EXE_FILE
    candidates={}
    for pc in range(lo,hi,4):
        r=record_at(exe,pc)
        if r: candidates[pc]=r
    groups=[]
    for pc in sorted(candidates):
        if not groups or pc-groups[-1][-1]!=0x1c: groups.append([pc])
        else: groups[-1].append(pc)
    anchor=0x800558A4
    main=next((g for g in groups if anchor in g),[])
    rows=[]
    if main:
        for idx,pc in enumerate(main):
            r=candidates[pc]
            rows.append({"group":"ANCHOR_TABLE","index":idx,"base_pc":h32(pc),"value_plus4":r["plus4"],
                         "pointer_plus8":h32(r["p1"]),"pointer_plusC":h32(r["p2"]),
                         "pointer_plus10":h32(r["p3"]),"name":r["name"],
                         "group_length":len(main),"stride":h32(0x1c)})
        start=main[0]; end=main[-1]+0x1c
    else: start=end=None
    snippet=[]
    if len(main)>=2:
        pc=main[0]; off=EXE_FILE+(pc-EXE_BASE); raw=exe[off:off+0x38]
        snippet=[{"image":"EXE","pc":h32(pc),"file_offset":h32(off),"bytes":hx(raw),"instruksi":"DATA_RECORD_PAIR","excerpt_id":"NA1_SCENE_TABLE_PAIR_56B"}]
    return rows,groups,(start,end),snippet,candidates,bool(main)

def scan_code_users(codes, valid, table_start, table_end, ptr_values):
    rows=[]; arithmetic=[]; low=table_start or EXE_BASE; high=table_end or (EXE_BASE+len(code)*4)
    for tag,code in codes:
        lo=EXE_BASE if tag=="EXE" else ENTRY_BASE
        for pc,i in sorted(code.items()):
            if pc not in valid[tag]: continue
            if i["op"] in (8,9,13) and i["rt"]!=0:
                v=const_reg(code,i["rt"],pc+4,lo)
                if v is not None and low<=v<high:
                    z=base_row(i,"NA1_TABLE_ADDRESS_CONSTANT"); z.update({"register":reg(i["rt"]),"constant":h32(v)}); rows.append(z)
            if i["op"] in LOADS:
                ea=eff_addr(code,i,lo); expr=sym_reg(code,i["rs"],pc,lo)
                cs=[int(x,16) for x in re.findall(r"0x([0-9A-Fa-f]{8})",expr)]
                table_ref=any(low<=x<high for x in cs)
                if (ea is not None and low-0x40<=ea<high+0x40) or table_ref:
                    z=base_row(i,"NA1_TABLE_LOAD_CANDIDATE")
                    z.update({"base_expression":expr,"effective_address":h32(ea) if ea is not None else "DYNAMIC",
                              "register_destination":reg(i["rt"]),"next_indirect_jump":"TIDAK TAHU",
                              "load_offset":i["imm"],"table_reference":"YES" if table_ref else "STATIC_EA_CANDIDATE"})
                    for q in range(pc+4,pc+0x81,4):
                        j=code.get(q)
                        if j and j["op"]==0 and j["fn"] in (8,9):
                            z["next_indirect_jump"]=h32(q)+" "+j["text"]+" target_expr="+sym_reg(code,j["rs"],q,lo); break
                    rows.append(z)
            if i["op"]==0 and i["fn"] in (0x18,0x19):
                a=const_reg(code,i["rs"],pc,lo); b=const_reg(code,i["rt"],pc,lo)
                if a==28 or b==28:
                    arithmetic.append(base_row(i,"SCALE_28_CONFIRMED")|{"scale_pattern":"MULT_BY_28",
                        "left_definition":sym_reg(code,i["rs"],pc,lo),"right_definition":sym_reg(code,i["rt"],pc,lo),
                        "input_expression":"literal multiplier 28"})
            elif i["op"]==0 and i["fn"]==0x23:
                dl=last_def(code,i["rs"],pc,lo); dr=last_def(code,i["rt"],pc,lo)
                if dl and dr and dl["op"]==0 and dr["op"]==0 and dl["fn"]==0 and dr["fn"]==0:
                    if dl["rs"]==0 and dr["rs"]==0 and dl["rt"]==dr["rt"] and {dl["sh"],dr["sh"]}=={2,5}:
                        arithmetic.append(base_row(i,"SCALE_28_CONFIRMED")|{"scale_pattern":"(index<<5)-(index<<2)",
                            "left_definition":dl["text"],"right_definition":dr["text"],"input_expression":reg(dl["rt"])})
            elif i["op"]==0 and i["fn"]==0x21:
                a=sym_reg(code,i["rs"],pc,lo); b=sym_reg(code,i["rt"],pc,lo)
                if any(("<<" + str(n)) in (a+b) for n in (2,3,4,5)):
                    arithmetic.append(base_row(i,"SCALE_ARITHMETIC_CANDIDATE")|{"scale_pattern":"TIDAK TAHU",
                        "left_definition":a,"right_definition":b,"input_expression":"TIDAK TAHU"})
    def dedup(xs):
        out={(x.get("image"),x.get("pc"),x.get("instruksi"),x.get("excerpt_id","")):x for x in xs}
        return list(out.values())
    return dedup(rows),dedup(arithmetic)

def pointer_words(blobs, targets):
    out=[]
    for tag,blob,base,filebase in [("EXE",blobs[0],EXE_BASE,EXE_FILE),("ENTRY1",blobs[1],ENTRY_BASE,ENTRY_FILE)]:
        for val in targets:
            needle=struct.pack("<I",val); pos=0
            while True:
                p=blob.find(needle,pos)
                if p<0: break
                if tag=="EXE":
                    if p>=EXE_FILE and p%4==0:
                        pc=EXE_BASE+(p-EXE_FILE)
                        out.append({"image":tag,"pc":h32(pc),"file_offset":h32(p),"bytes":hx(blob[p:p+4]),
                                    "instruksi":"DATA_POINTER_WORD","value":h32(val),"classification":"POINTER_WORD"})
                else:
                    if p%4==0:
                        pc=ENTRY_BASE+p
                        out.append({"image":tag,"pc":h32(pc),"file_offset":h32(ENTRY_FILE+p),"bytes":hx(blob[p:p+4]),
                                    "instruksi":"DATA_POINTER_WORD","value":h32(val),"classification":"POINTER_WORD"})
                pos=p+1
    return out

def source_value_class(expr, value):
    if value is not None: return "KONSTANTA"
    if re.fullmatch(r"MEM(?:8|16|32|8S|16S)\[[^\]]+\]",expr): return "SALINAN_DARI_MEMORI"
    if expr.startswith("INCOMING_"): return "ARGUMEN_MASUK"
    if "MEM" in expr and any(op in expr for op in ("+","-","&","|")): return "TRANSFORMASI_MEMORI_TIDAK_TERIDENTIFIKASI"
    return "TIDAK TAHU"

def global_writers(codes, valid, globals_):
    rows=[]
    for tag,code in codes:
        lo=EXE_BASE if tag=="EXE" else ENTRY_BASE
        for pc,i in sorted(code.items()):
            if pc not in valid[tag] or i["op"] not in STORES: continue
            ea=eff_addr(code,i,lo)
            if ea is None: continue
            width=WIDTH[i["op"]]
            for g in globals_:
                if ea<=g<ea+width:
                    val=const_reg(code,i["rt"],pc,lo); expr=sym_reg(code,i["rt"],pc,lo)
                    mask=(1<<(8*width))-1
                    stored=(val & mask) if val is not None else None
                    z=base_row(i,"NA2_GLOBAL_WRITER_"+h32(g))
                    z.update({"target_global":h32(g),"store_width":width,"effective_address":h32(ea),
                              "source_register":reg(i["rt"]),
                              "source_constant":h32(val) if val is not None else "TIDAK TAHU",
                              "stored_constant":(h32(stored) if width==4 else f"0x{stored:0{width*2}X}") if stored is not None else "TIDAK TAHU",
                              "source_expression":expr,"source_class":source_value_class(expr,val)})
                    rows.append(z)
    return rows

def matrix_writers(codes, valid):
    rows=[]; partial=[]; all_sb=[]
    anchors=("0x8013EA74","0x8013EB1C","0x8013EA6E")
    for tag,code in codes:
        lo=EXE_BASE if tag=="EXE" else ENTRY_BASE
        for pc,i in sorted(code.items()):
            if pc not in valid[tag] or i["op"]!=0x28: continue
            b=sym_reg(code,i["rs"],pc,lo); src=sym_reg(code,i["rt"],pc,lo)
            base_const=eff_addr(code,i,lo); val=const_reg(code,i["rt"],pc,lo)
            src_class=source_value_class(src,val)
            n=sum(a in b.upper() for a in anchors)
            census=base_row(i,"NA2_ALL_SB_STORE")
            census.update({"base_expression":b,"effective_address":h32(base_const) if base_const is not None else "TIDAK TAHU",
                           "base_class":"STATIC_ADDRESS" if base_const is not None else "DYNAMIC_OR_ALIAS_UNRESOLVED",
                           "source_expression":src,"source_class":src_class,"matrix_anchor_count":n})
            all_sb.append(census)
            if n>=1:
                item=base_row(i,"NA2_MATRIX_WRITER_CANDIDATE")
                if val is not None: cls="KONSTANTA"
                elif re.fullmatch(r"MEM(?:8|16|32|8S|16S)\[[^\]]+\]",src): cls="SALINAN_DARI_MEMORI"
                elif "MEM" in src and "-" in src: cls="SELISIH_KANDIDAT_BELUM_TERVERIFIKASI_LEVEL"
                else: cls="TIDAK TAHU"
                item.update({"base_expression":b,"source_expression":src,"source_class":cls,"anchor_count":n})
                (rows if n==3 else partial).append(item)
    return rows,partial,all_sb

def caller_rows(code, valid, refs, target):
    out=[]
    for pc,i in sorted(code.items()):
        if pc not in valid: continue
        t=direct_target(i)
        if t==target:
            z=base_row(i); z.update({"target":h32(target),"kind":"JAL" if i["op"]==3 else "J","origin":"DIRECT"})
            out.append(z)
        if i["op"]==0 and i["fn"] in (8,9):
            v=const_reg(code,i["rs"],pc,ENTRY_BASE)
            if v==target:
                z=base_row(i); z.update({"target":h32(target),"kind":"JALR" if i["fn"]==9 else "JR","origin":"STATIC_REGISTER"})
                out.append(z)
    for x in refs.get(target,[]):
        z={"image":"ENTRY1","pc":h32(x["pc"]),"file_offset":h32(ENTRY_FILE+x["pc"]-ENTRY_BASE),
           "bytes":"TIDAK TAHU","instruksi":"GHIDRA_REFERENCE","target":h32(target),
           "kind":x["kind"],"origin":"GHIDRA_XREF caller="+x["caller_name"]+" entry="+x["caller_entry"]}
        i=code.get(x["pc"])
        if i: z.update({"bytes":hx(i["raw"]),"instruksi":i["text"]})
        out.append(z)
    unique={(x.get("pc"),x.get("kind"),x.get("origin")):x for x in out}
    return list(unique.values())

def validate_outputs(exe, entry):
    rows=[]; checked=decoded=fail=0
    for path in sorted(OUT.glob("NA*.tsv")):
        try: data=list(csv.DictReader(path.open(encoding="utf-8"),delimiter="\t"))
        except Exception: continue
        for n,r in enumerate(data,2):
            if not all(r.get(k) for k in ("image","pc","file_offset","bytes")): continue
            if r["bytes"]=="TIDAK TAHU": continue
            try:
                pc=int(r["pc"],16); fo=int(r["file_offset"],16)
                raw=bytes(int(x,16) for x in r["bytes"].split())
                tag=r["image"]
                expected=(EXE_FILE+pc-EXE_BASE) if tag=="EXE" else (ENTRY_FILE+pc-ENTRY_BASE)
                source=exe[fo:fo+len(raw)] if tag=="EXE" else entry[fo-ENTRY_FILE:fo-ENTRY_FILE+len(raw)]
                ok=(fo==expected and source==raw and len(raw)>0)
                dec="NOT_CODE"
                if ok and len(raw)==4 and not r.get("instruksi","").startswith("DATA_"):
                    ins=next(md.disasm(raw,pc,count=1),None)
                    ok=ins is not None
                    dec="PASS" if ins else "FAIL"
                    if ins: decoded+=1
                checked+=1
                if not ok: fail+=1
                rows.append({"file":path.name,"line":n,"image":tag,"pc":r["pc"],"file_offset":r["file_offset"],
                             "raw_check":"PASS" if ok else "FAIL","decode_check":dec})
            except Exception as e:
                checked+=1; fail+=1
                rows.append({"file":path.name,"line":n,"image":r.get("image"),"pc":r.get("pc"),
                             "file_offset":r.get("file_offset"),"raw_check":"FAIL","decode_check":"ERROR:"+type(e).__name__})
    write_tsv("NA0_FILE_OFFSET_DECODE_VALIDATION.tsv",rows,["file","line","image","pc","file_offset","raw_check","decode_check"])
    return checked,decoded,fail

def excerpt_audit():
    seen={}; sources=[]; root=Path("reports/stage1_economy")
    for prefix in ("ja-run-","ka-run-","la-run-","ma-run-","na-run-"):
        sources.extend(root.glob(prefix+"*/attempt-*/*.tsv"))
    sources.extend(OUT.glob("NA*.tsv"))
    for path in sources:
        try: rr=csv.DictReader(path.open(encoding="utf-8"),delimiter="\t")
        except Exception: continue
        if not rr.fieldnames or not {"image","pc","bytes"}.issubset(rr.fieldnames): continue
        for row in rr:
            raw=row.get("bytes","")
            if not raw or raw=="TIDAK TAHU": continue
            try: pc=int(row["pc"],16); b=bytes(int(x,16) for x in raw.split())
            except Exception: continue
            if not b or len(b)%4: continue
            key=(row["image"],pc)
            if key not in seen:
                seen[key]={"bytes":b,"files":{str(path)}}
            elif seen[key]["bytes"]==b"CONFLICT":
                seen[key]["files"].add(str(path))
            elif seen[key]["bytes"].startswith(b):
                seen[key]["files"].add(str(path))
            elif b.startswith(seen[key]["bytes"]):
                seen[key]["bytes"]=b
                seen[key]["files"].add(str(path))
            elif seen[key]["bytes"]!=b:
                seen[key]["bytes"]=b"CONFLICT"
                seen[key]["files"].add(str(path))
            else:
                seen[key]["files"].add(str(path))
    spans=[]
    for image in sorted(set(k[0] for k in seen)):
        intervals=sorted((pc,pc+len(v["bytes"]),v["files"]) for (im,pc),v in seen.items()
                         if im==image and v["bytes"]!=b"CONFLICT")
        cur=None
        for start,end,files in intervals:
            if cur is None or start>cur["end"]:
                if cur is not None: spans.append(cur)
                cur={"image":image,"start":start,"end":end,"rows":1,"files":set(files)}
            else:
                cur["end"]=max(cur["end"],end); cur["rows"]+=1; cur["files"].update(files)
        if cur is not None: spans.append(cur)
    detailed=[]
    for x in spans:
        detailed.append({"image":x["image"],"start_pc":h32(x["start"]),"end_pc_exclusive":h32(x["end"]),
                         "contiguous_bytes":x["end"]-x["start"],"unique_rows":x["rows"],
                         "source_files":";".join(sorted(x["files"]))})
    write_tsv("NA0_AGGREGATE_EXCERPT_SPANS.tsv",detailed,["image","start_pc","end_pc_exclusive","contiguous_bytes","unique_rows","source_files"])
    maximum=max((x["contiguous_bytes"] for x in detailed),default=0)
    offenders=[x for x in detailed if x["contiguous_bytes"]>64]
    conflicts=sum(1 for v in seen.values() if v["bytes"]==b"CONFLICT")
    write_txt("NA0_AGGREGATE_EXCERPT_AUDIT.txt",
        "AGGREGATION=union of byte intervals by image; unique image+PC rows across every ja-*, ka-*, la-*, ma-*, na-* TSV\n"+
        f"UNIQUE_IMAGE_PC_ROWS={len(seen)}\nMAX_CONTIGUOUS_BYTES={maximum}\nLIMIT_BYTES=64\n"+
        "STATUS="+("PASS" if not offenders and conflicts==0 else "FAIL")+"\n"+
        "OVER_LIMIT="+(";".join(f'{x["image"]}:{x["start_pc"]}..{h32(int(x["end_pc_exclusive"],16)-1)}={x["contiguous_bytes"]}B' for x in offenders) or "none")+"\n"+
        "CONFLICTING_DUPLICATE_ADDRESSES="+str(conflicts)+"\n"+
        "OFFENDER_PROVENANCE="+(";".join(f'{x["image"]}:{x["start_pc"]}:{x["source_files"]}' for x in offenders) or "none")+"\n")
    return maximum,len(offenders)

def actions_census():
    token=os.environ.get("GH_TOKEN","")
    page=1; total=0; allruns=[]
    while True:
        url=f"https://api.github.com/repos/{REPO}/actions/runs?per_page=100&page={page}"
        req=urllib.request.Request(url,headers={"Authorization":"Bearer "+token,"Accept":"application/vnd.github+json","X-GitHub-Api-Version":"2022-11-28"})
        with urllib.request.urlopen(req,timeout=90) as r: j=json.loads(r.read().decode())
        total=int(j.get("total_count",0)); rs=j.get("workflow_runs",[]); allruns.extend(rs)
        if not rs or len(allruns)>=total: break
        page+=1
        if page>100: raise RuntimeError("ACTIONS_PAGINATION_LIMIT")
    rows=[{"run_id":x.get("id"),"event":x.get("event"),"status":x.get("status"),"conclusion":x.get("conclusion") or "TIDAK TAHU",
           "workflow":x.get("path"),"head_sha":x.get("head_sha"),"created_at":x.get("created_at")} for x in allruns]
    write_tsv("NA0_ACTIONS_ALL_EVENTS.tsv",rows,["run_id","event","status","conclusion","workflow","head_sha","created_at"])
    return total,len(rows),page,sum(x.get("event")=="schedule" for x in allruns),sum(x.get("status")!="completed" for x in allruns)

def ledger():

    rows=[]
    for line in raw:

def na_window(code,start,count,label):
    out=[]
    for n in range(count):
        i=code.get(start+n*4)
        if i: out.append(base_row(i,label))
    return out

def na_edge_rows(codes):
    rows=[]
    for tag,code in codes:
        lo=EXE_BASE if tag=="EXE" else ENTRY_BASE
        for pc,i in sorted(code.items()):
            t=None; kind=""
            if i["op"] in (1,4,5,6,7):
                t=(pc+4+(i["imm"]<<2))&0xffffffff
                kind={1:"REGIMM",4:"BEQ",5:"BNE",6:"BLEZ",7:"BGTZ"}[i["op"]]
            elif i["op"] in (2,3):
                t=direct_target(i); kind="JAL" if i["op"]==3 else "J"
            elif i["op"]==0 and i["fn"] in (8,9):
                t=const_reg(code,i["rs"],pc,lo); kind="JALR" if i["fn"]==9 else "JR"
            if kind: rows.append((tag,pc,i,t,kind))
    return rows

def na_call_arg(code,pc,r):
    q=code.get(pc+4)
    if q and dest(q)==r:
        ex=sym_reg(code,r,pc+8,max(ENTRY_BASE,pc-0x40))
        v=const_reg(code,r,pc+8,max(ENTRY_BASE,pc-0x40))
        return ("KONSTANTA" if v is not None else "SALINAN" if ex.startswith(("INCOMING_","MEM")) else "TIDAK TAHU",
                h32(q["pc"]),ex,"DELAY_SLOT")
    floor=pc-0x40
    hit=None
    for p in range(pc-4,floor-1,-4):
        i=code.get(p)
        if not i: continue
        if dest(i)==r: hit=i; break
        if i["op"] in (1,2,3,4,5,6,7) or (i["op"]==0 and i["fn"] in (8,9)):
            return "TIDAK TAHU","TIDAK TAHU","TIDAK TAHU: control-flow boundary","CONTROL_FLOW_BOUNDARY"
    if hit is None: return "ARGUMEN MASUK","TIDAK TAHU","INCOMING_"+reg(r),"NO_LOCAL_DEF_IN_16"
    ex=sym_reg(code,r,pc,floor); v=const_reg(code,r,pc,floor)
    if v is not None: cls="KONSTANTA"
    elif hit["op"] in LOADS or (hit["op"]==0 and hit["fn"] in (0x21,0x25) and (hit["rs"]==0 or hit["rt"]==0)): cls="SALINAN"
    else: cls="TIDAK TAHU"
    return cls,h32(hit["pc"]),ex,"DEFINITION_IN_16"

def na_matrix_uses(code,load,lo,hi):
    # Linear taint from the pointer loaded through global M. Also retain the
    # complete symbolic base expression and direct JAL target for each use.
    aliases={load["rt"]:"M_PTR@"+h32(load["pc"])}; rows=[]
    for p in range(load["pc"]+4,min(hi,load["pc"]+0x180)+1,4):
        i=code.get(p)
        if not i: continue
        op,fn=i["op"],i["fn"]
        base_expr=sym_reg(code,i["rs"],p,lo)
        addr_expr="("+base_expr+f"{i['imm']:+d})"
        if op in LOADS and i["rs"] in aliases:
            val_expr=sym_reg(code,i["rt"],p+4,lo)
            rows.append({"load_pc":h32(load["pc"]),"use_pc":h32(p),"file_offset":h32(i["fileoff"]),
                "instruction":i["text"],"use_kind":"LOAD_FROM_M_DERIVED_POINTER",
                "address_expression":addr_expr,"derived_pointer_expression":aliases[i["rs"]]+f"{i['imm']:+d}",
                "source_expression":val_expr,"source_class":source_value_class(val_expr,None),
                "result_register":reg(i["rt"]),"jal_target":"TIDAK TAHU","jal_arg_registers":"TIDAK TAHU"})
        if op in STORES and i["rs"] in aliases:
            v=const_reg(code,i["rt"],p,lo); ex=sym_reg(code,i["rt"],p,lo)
            rows.append({"load_pc":h32(load["pc"]),"use_pc":h32(p),"file_offset":h32(i["fileoff"]),
                "instruction":i["text"],"use_kind":"STORE_THROUGH_M_DERIVED_POINTER",
                "address_expression":addr_expr,"derived_pointer_expression":aliases[i["rs"]]+f"{i['imm']:+d}",
                "source_expression":ex,"source_class":source_value_class(ex,v),
                "result_register":"TIDAK TAHU","jal_target":"TIDAK TAHU","jal_arg_registers":"TIDAK TAHU"})
        if op==3 and any(r in aliases for r in (4,5,6,7)):
            args=[r for r in (4,5,6,7) if r in aliases]
            rows.append({"load_pc":h32(load["pc"]),"use_pc":h32(p),"file_offset":h32(i["fileoff"]),
                "instruction":i["text"],"use_kind":"M_POINTER_PASSED_TO_JAL",
                "address_expression":";".join(aliases[r] for r in args),
                "derived_pointer_expression":";".join(reg(r) for r in args),
                "source_expression":"TIDAK TAHU","source_class":"TIDAK TAHU",
                "result_register":"TIDAK TAHU","jal_target":h32(direct_target(i)),
                "jal_arg_registers":";".join(reg(r) for r in args)})
        d=dest(i)
        if d is not None:
            new=None
            if op in (8,9) and i["rs"] in aliases: new=aliases[i["rs"]]+f"{i['imm']:+d}"
            elif op==0 and fn in (0x21,0x25) and (i["rs"] in aliases or i["rt"] in aliases):
                new=(aliases.get(i["rs"],"")+"+"+aliases.get(i["rt"],"")).strip("+")
            aliases.pop(d,None)
            if new is not None: aliases[d]=new
        if op in (1,2,3,4,5,6,7) or (op==0 and fn in (8,9)): break
    return rows
